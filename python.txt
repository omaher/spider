#Execute a shell command from Python
import subprocess
def run_command(command):
    result = subprocess.run(command, shell=True, capture_output=True, text=True)
    return result.stdout

#Monitor a URL’s response code using Python
import requests

def check_url_status(url):
    try:
        response = requests.get(url)
        print(f"Status Code: {response.status_code}")
    except requests.RequestException as e:
        print(f"Error: {e}")

check_url_status("https://example.com")

####### How many times no. repeated ####################

numbers = [1, 2, 3, 2, 4, 1, 5, 2, 3]

repeats = {}

for num in numbers:
    if num in repeats:
        repeats[num] += 1
    else:
        repeats[num] = 1

# Print only repeated numbers
for num, count in repeats.items():
    if count > 1:
        print(f"{num} is repeated {count} times")
